# Procedural alpha textures for BLINK overgrowth. Usage: python gen_textures.py <outDir>
import sys, os, math, random
import numpy as np
from PIL import Image, ImageDraw, ImageFilter

out = sys.argv[1]
os.makedirs(out, exist_ok=True)
SS = 2  # supersample


def finish(img, name, size, bleed=None):
    """Downsample, then bleed colour into transparent pixels so mips don't get dark fringes."""
    img = img.resize(size, Image.LANCZOS)
    a = np.asarray(img).astype(np.float32)
    rgb, al = a[..., :3], a[..., 3:4] / 255.0
    if bleed is None:
        w = al.sum()
        bleed = (rgb * al).reshape(-1, 3).sum(0) / max(w, 1)
    # blur the coloured pixels outward a few times
    col = Image.fromarray(np.clip(rgb * al + np.array(bleed) * (1 - al), 0, 255).astype(np.uint8))
    blurred = np.asarray(col.filter(ImageFilter.GaussianBlur(4))).astype(np.float32)
    rgb2 = rgb * al + blurred * (1 - al)
    res = np.concatenate([rgb2, al * 255], -1)
    Image.fromarray(np.clip(res, 0, 255).astype(np.uint8), 'RGBA').save(os.path.join(out, name))
    print('wrote', name)


def jitter(c, r, v=18):
    return tuple(int(max(0, min(255, x + r.randint(-v, v)))) for x in c) + (255,)


def ivy_leaf(d, cx, cy, s, ang, col):
    pts = []
    for i in range(40):
        t = i / 40 * 2 * math.pi
        rr = s * (0.75 + 0.3 * abs(math.cos(2.5 * t)))  # 5-lobed
        x, y = rr * math.cos(t), rr * math.sin(t) * 0.9
        ca, sa = math.cos(ang), math.sin(ang)
        pts.append((cx + x * ca - y * sa, cy + x * sa + y * ca))
    d.polygon(pts, fill=col)
    # midrib
    d.line([(cx, cy), (cx + s * 0.8 * math.cos(ang), cy + s * 0.8 * math.sin(ang))], fill=tuple(int(c * 0.7) for c in col[:3]) + (255,), width=max(1, int(s / 10)))


# ---------------------------------------------------------------- Ivy (wall patch, tiles vertically)
def ivy(name, w, h, density, seed, hanging=False):
    r = random.Random(seed)
    W, H = w * SS, h * SS
    img = Image.new('RGBA', (W, H), (0, 0, 0, 0))
    d = ImageDraw.Draw(img)
    stems = 3 if hanging else 7
    greens = [(38, 62, 24), (52, 78, 30), (30, 50, 22), (64, 82, 36), (46, 58, 28)]
    for s in range(stems):
        x = r.uniform(0.15, 0.85) * W
        y = 0
        pts = [(x, y)]
        while y < H:
            y += r.uniform(20, 40) * SS
            x += r.uniform(-14, 14) * SS
            x = min(max(x, 10), W - 10)
            pts.append((x, y))
        d.line(pts, fill=(58, 44, 30, 255), width=3 * SS)
        for (px, py) in pts:
            n = density
            for _ in range(n):
                lx = px + r.uniform(-30, 30) * SS * (0.4 if hanging else 1)
                ly = py + r.uniform(-20, 20) * SS
                sz = r.uniform(9, 17) * SS * (1 - 0.35 * (py / H) if hanging else 1)
                ivy_leaf(d, lx, ly, sz, r.uniform(0, 2 * math.pi), jitter(r.choice(greens), r))
    finish(img, name, (w, h))


# ---------------------------------------------------------------- Grass tuft
def grass(name, seed):
    r = random.Random(seed)
    W, H = 512 * SS, 512 * SS
    img = Image.new('RGBA', (W, H), (0, 0, 0, 0))
    d = ImageDraw.Draw(img)
    for _ in range(70):
        bx = r.uniform(0.1, 0.9) * W
        hgt = r.uniform(0.35, 0.95) * H
        bend = r.uniform(-0.25, 0.25) * W
        wid = r.uniform(5, 11) * SS
        dry = r.random() < 0.35
        col = jitter((120, 112, 60) if dry else (58, 86, 34), r, 14)
        pts_l, pts_r = [], []
        for i in range(11):
            t = i / 10
            x = bx + bend * t * t
            y = H - hgt * t
            ww = wid * (1 - t)
            pts_l.append((x - ww / 2, y)); pts_r.append((x + ww / 2, y))
        d.polygon(pts_l + pts_r[::-1], fill=col)
    finish(img, name, (256, 256))


# ---------------------------------------------------------------- Pine branch (card: u along branch, v across)
def pine_branch(name, seed):
    r = random.Random(seed)
    W, H = 512 * SS, 256 * SS
    img = Image.new('RGBA', (W, H), (0, 0, 0, 0))
    d = ImageDraw.Draw(img)
    def stem(x0, y0, ang, length, width, depth):
        pts = [(x0, y0)]
        x, y = x0, y0
        steps = 24
        for i in range(steps):
            t = (i + 1) / steps
            x += math.cos(ang) * length / steps
            y += math.sin(ang) * length / steps
            pts.append((x, y))
            # needles
            spread = (1 - t * 0.6) * (H * 0.22 if depth == 0 else H * 0.1)
            for side in (-1, 1):
                for _ in range(6):
                    na = ang + side * r.uniform(0.6, 1.2)
                    nl = spread * r.uniform(0.5, 1.0)
                    col = jitter((24, 46, 32), r, 10)
                    d.line([(x, y), (x + math.cos(na) * nl, y + math.sin(na) * nl)], fill=col, width=3 * SS)
            if depth == 0 and i % 5 == 3 and t < 0.85:
                for side in (-1, 1):
                    stem(x, y, ang + side * r.uniform(0.5, 0.8), length * (1 - t) * 0.5, width * 0.6, 1)
        d.line(pts, fill=(52, 38, 28, 255), width=int(width))
    stem(4 * SS, H / 2, 0.0, W * 0.96, 5 * SS, 0)
    finish(img, name, (256, 128))


# ---------------------------------------------------------------- Leaf cluster (deciduous canopy card)
def leaf_cluster(name, seed, palette, w=512):
    r = random.Random(seed)
    W = w * SS
    img = Image.new('RGBA', (W, W), (0, 0, 0, 0))
    d = ImageDraw.Draw(img)
    for _ in range(420):
        a = r.uniform(0, 2 * math.pi)
        rr = (r.random() ** 0.6) * W * 0.46
        cx, cy = W / 2 + math.cos(a) * rr, W / 2 + math.sin(a) * rr
        s = r.uniform(9, 16) * SS
        ang = r.uniform(0, 2 * math.pi)
        col = jitter(r.choice(palette), r, 12)
        # ellipse leaf as polygon
        pts = []
        for i in range(16):
            t = i / 16 * 2 * math.pi
            x, y = math.cos(t) * s, math.sin(t) * s * 0.45
            pts.append((cx + x * math.cos(ang) - y * math.sin(ang), cy + x * math.sin(ang) + y * math.cos(ang)))
        d.polygon(pts, fill=col)
    finish(img, name, (w, w))


# ---------------------------------------------------------------- Leaf litter (ground decal)
def litter(name, seed):
    r = random.Random(seed)
    W = 512 * SS
    img = Image.new('RGBA', (W, W), (0, 0, 0, 0))
    d = ImageDraw.Draw(img)
    pal = [(92, 58, 26), (120, 76, 30), (70, 46, 24), (104, 88, 40), (60, 40, 22), (130, 60, 24)]
    for _ in range(260):
        a = r.uniform(0, 2 * math.pi)
        rr = (r.random() ** 0.8) * W * 0.47
        cx, cy = W / 2 + math.cos(a) * rr, W / 2 + math.sin(a) * rr
        s = r.uniform(10, 20) * SS
        ang = r.uniform(0, 2 * math.pi)
        col = jitter(r.choice(pal), r, 10)
        pts = []
        for i in range(18):
            t = i / 18 * 2 * math.pi
            x, y = math.cos(t) * s, math.sin(t) * s * (0.5 + 0.15 * math.sin(3 * t))
            pts.append((cx + x * math.cos(ang) - y * math.sin(ang), cy + x * math.sin(ang) + y * math.cos(ang)))
        d.polygon(pts, fill=col)
        d.line([(cx - math.cos(ang) * s, cy - math.sin(ang) * s), (cx + math.cos(ang) * s, cy + math.sin(ang) * s)],
               fill=tuple(int(c * 0.6) for c in col[:3]) + (255,), width=SS)
    finish(img, name, (512, 512))


# ---------------------------------------------------------------- Treeline silhouette (tiles horizontally)
def treeline(name, seed, w=2048, h=512):
    r = random.Random(seed)
    W, H = w * SS, h * SS
    img = Image.new('RGBA', (W, H), (0, 0, 0, 0))
    d = ImageDraw.Draw(img)
    base_col = (14, 20, 18, 255)
    d.rectangle([0, H * 0.82, W, H], fill=base_col)
    trees = []
    x = 0
    while x < W:
        trees.append((x, r.uniform(0.35, 0.95) * H, r.uniform(0.16, 0.26) * H))
        x += r.uniform(0.01, 0.03) * W
    trees.sort(key=lambda t: t[1])
    for (tx, th, tw) in trees:
        for off in (-W, 0, W):  # wrap
            cx = tx + off
            top = H - th
            tiers = 9
            for i in range(tiers):
                t = i / tiers
                y0 = top + (H - top) * t
                y1 = top + (H - top) * min(1, t + 0.22)
                half = tw * (0.15 + 0.85 * (t + 0.1)) * r.uniform(0.85, 1.15)
                d.polygon([(cx, y0), (cx - half, y1), (cx + half, y1)], fill=base_col)
            d.rectangle([cx - tw * 0.06, top, cx + tw * 0.06, H], fill=base_col)
    finish(img, name, (w, h), bleed=(14, 20, 18))


# ---------------------------------------------------------------- Cobweb (corner at top-left of the card)
def cobweb(name, seed):
    r = random.Random(seed)
    W = 512 * SS
    img = Image.new('RGBA', (W, W), (0, 0, 0, 0))
    d = ImageDraw.Draw(img)
    col = (200, 200, 195, 255)
    spokes = 9
    angs = [math.pi / 2 * (i / (spokes - 1)) * 0.96 + 0.02 + r.uniform(-0.04, 0.04) for i in range(spokes)]
    lens = [W * r.uniform(0.75, 1.0) for _ in angs]
    for a, l in zip(angs, lens):
        d.line([(0, 0), (math.cos(a) * l, math.sin(a) * l)], fill=col, width=3 * SS // 2)
    ring = W * 0.08
    while ring < W * 0.9:
        for i in range(spokes - 1):
            a0, a1 = angs[i], angs[i + 1]
            if ring > min(lens[i], lens[i + 1]):
                continue
            p0 = (math.cos(a0) * ring, math.sin(a0) * ring)
            p1 = (math.cos(a1) * ring, math.sin(a1) * ring)
            mid = ((p0[0] + p1[0]) / 2, (p0[1] + p1[1]) / 2)
            sag = 0.88  # pulled toward the corner
            m = (mid[0] * sag, mid[1] * sag)
            d.line([p0, m, p1], fill=col, width=SS + 1)
        ring *= r.uniform(1.18, 1.3)
    # a few torn strands
    for _ in range(5):
        a = r.uniform(0.1, 1.4)
        l0 = W * r.uniform(0.2, 0.6)
        x0, y0 = math.cos(a) * l0, math.sin(a) * l0
        d.line([(x0, y0), (x0 + r.uniform(-30, 30) * SS, y0 + r.uniform(40, 120) * SS)], fill=col, width=SS)
    finish(img, name, (512, 512), bleed=(200, 200, 195))


# ---------------------------------------------------------------- Broken floorboards (dark gaps, visual-only hole)
def floor_hole(name, seed):
    r = random.Random(seed)
    W = 512 * SS
    img = Image.new('RGBA', (W, W), (0, 0, 0, 0))
    d = ImageDraw.Draw(img)
    plank = W / 5
    for i in range(5):
        if r.random() < 0.25:
            continue
        x0 = i * plank + 3 * SS
        x1 = (i + 1) * plank - 3 * SS
        cy = W / 2 + r.uniform(-0.1, 0.1) * W
        half = r.uniform(0.18, 0.42) * W
        top = []
        bot = []
        for k in range(9):
            x = x0 + (x1 - x0) * k / 8
            top.append((x, cy - half + r.uniform(-28, 28) * SS))
            bot.append((x, cy + half + r.uniform(-28, 28) * SS))
        d.polygon(top + bot[::-1], fill=(8, 6, 5, 255))
        # splintered lighter edge
        d.line(top, fill=(70, 50, 34, 255), width=3 * SS)
        d.line(bot, fill=(70, 50, 34, 255), width=3 * SS)
    finish(img, name, (512, 512))


# ---------------------------------------------------------------- Tileable noise: R low, G mid, B high, A vertical streaks
def noise(name, n=256):
    rs = np.random.RandomState(7)
    fx = np.fft.fftfreq(n)[:, None]
    fy = np.fft.fftfreq(n)[None, :]
    def band(f0, sx=1.0, sy=1.0):
        w = rs.randn(n, n)
        f = np.sqrt((fx * sx) ** 2 + (fy * sy) ** 2)
        k = np.exp(-(f / f0) ** 2)
        v = np.real(np.fft.ifft2(np.fft.fft2(w) * k))
        v = (v - v.mean()) / (v.std() * 4) + 0.5
        return np.clip(v, 0, 1)
    ch = [band(4 / n), band(16 / n), band(48 / n), band(24 / n, sx=0.12, sy=1.0)]  # A: stretched along rows -> transposed below
    ch[3] = ch[3].T  # streaks run vertically (along v)
    img = np.stack(ch, -1) * 255
    Image.fromarray(img.astype(np.uint8), 'RGBA').save(os.path.join(out, name))
    print('wrote', name)
